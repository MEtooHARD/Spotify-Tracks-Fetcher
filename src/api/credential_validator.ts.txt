import { GetToken } from "./auth";
import { try_catch } from "../utils/wrapper";

const TRACK_ID = "53iuhJlwXhSER5J2IYYv1W";

/**
 * 驗證 credential 的實際狀態
 * @returns true: 可用 | false: 有錯誤 | number: 被 ban 的剩餘秒數
 */
export async function validateCredential(
    clientID: string,
    secret: string
): Promise<boolean | number> {
    try {
        // 嘗試取得 token
        const [tokenRes, tokenErr] = await GetToken(clientID, secret);

        if (tokenErr) {
            return false;
        }

        const token = tokenRes.access_token;

        // 等待一下讓 Spotify API 更新 token 狀態
        await new Promise(resolve => setTimeout(resolve, 3000));

        // 用 token 測試 API（用 tracks 而不是 /me）
        const [apiRes, apiErr] = await try_catch(
            fetch(`https://api.spotify.com/v1/tracks/${TRACK_ID}`, {
                headers: { 'Authorization': `Bearer ${token}` }
            })
        );

        if (apiErr) {
            return false;
        }

        const status = apiRes!.status;

        if (status === 200) {
            return true;
        } else if (status === 429) {
            const retryAfter = apiRes!.headers.get('retry-after');
            const retrySeconds = parseInt(retryAfter || '3600');
            return retrySeconds;
        } else {
            return false;
        }
    } catch (error) {
        return false;
    }
}
